### `Pose2d.java`
```java
package org.firstinspires.ftc.teamcode.configs;

import com.example.instantauto.configs.MetaField;
import com.acmerobotics.roadrunner.Pose2d;

public class Pose2d implements MetaField<Pose2d> {
    public final double x, y, heading;

    public Pose2d(double x, double y, double heading) {
        this.x = x; this.y = y; this.heading = heading;
    }

    @Override public String getIdentifier() { return "pose2d"; }
    @Override public Class<?>[] getParamTypes() { 
        return new Class[]{double.class, double.class, double.class}; 
    }

    public Pose2d getRRPose2d() {
        return new Pose2d(x, y, Math.toRadians(heading));
    }
}
```

## 2. ConfigManager.java

```java
package org.firstinspires.ftc.teamcode.configs;

import com.example.instantauto.configs.MetaFieldRegistry;
import com.example.instantauto.actions.UserActionRegistry;
import com.qualcomm.robotcore.eventloop.opmode.OpMode;
import com.qualcomm.robotcore.hardware.DistanceSensor;
import com.qualcomm.robotcore.hardware.HardwareMap;
import com.qualcomm.robotcore.hardware.VoltageSensor;
import org.firstinspires.ftc.robotcore.external.navigation.DistanceUnit;

public class ConfigManager {
    public static void init(OpMode opMode) {
        HardwareMap hw = opMode.hardwareMap;

        // 1. Register MetaField types
        MetaFieldRegistry.registerType(new Pose2d(0, 0, 0));
        MetaFieldRegistry.registerType(new IntakeSetting("", false, 0));

        // 2. Static fields with defaults
        MetaFieldRegistry.registerField("Starting", Pose2d.class, new Pose2d(0, 0, 0));
        MetaFieldRegistry.registerField("Title", String.class, "");
        MetaFieldRegistry.registerField("isBlue", Boolean.class, true);
        MetaFieldRegistry.registerField("hasPreload", Boolean.class, false);
        
        // Poses
        MetaFieldRegistry.registerField("blueScorePose", Pose2d.class, new Pose2d(48, 24, 90));
        MetaFieldRegistry.registerField("redScorePose", Pose2d.class, new Pose2d(48, -24, -90));
        MetaFieldRegistry.registerField("blueIntakePose", Pose2d.class, new Pose2d(-36, 60, 180));
        MetaFieldRegistry.registerField("redIntakePose", Pose2d.class, new Pose2d(-36, -60, 180));
        MetaFieldRegistry.registerField("parkPose", Pose2d.class, new Pose2d(12, 12, 0));
        
        // Tuning
        MetaFieldRegistry.registerField("maxVelocity", Double.class, 60.0);
        MetaFieldRegistry.registerField("intakePower", Double.class, 1.0);
        MetaFieldRegistry.registerField("scoreWaitTime", Double.class, 0.5);

        // 3. Sensor suppliers (live values)
        MetaFieldRegistry.registerField("distanceCm", Double.class,
            () -> hw.get(DistanceSensor.class, "distance").getDistance(DistanceUnit.CM));
        MetaFieldRegistry.registerField("batteryVoltage", Double.class,
            () -> hw.voltageSensor.iterator().next().getVoltage());

        // 4. Condition suppliers (unchangeable logic)
        UserActionRegistry.registerCondition("sampleNear", () ->
            hw.get(DistanceSensor.class, "distance").getDistance(DistanceUnit.CM) < 15.0);
        UserActionRegistry.registerCondition("lowBattery", () ->
            hw.voltageSensor.iterator().next().getVoltage() < 11.0);
        UserActionRegistry.registerCondition("isActive", () -> true);
    }
}
```

## 3. Subsystems

### `Intake.java`
```java
package org.firstinspires.ftc.teamcode.subsystems;

import com.example.instantauto.actions.Action;
import com.qualcomm.robotcore.hardware.HardwareMap;
import com.qualcomm.robotcore.hardware.Motor;
import com.qualcomm.robotcore.hardware.Servo;

public class Intake {
    private Motor motor;
    private Servo servo;

    public static final double INTAKE_POWER = 1.0;
    public static final double OUTTAKE_POWER = -1.0;
    public static final double SERVO_OPEN = 0.5;
    public static final double SERVO_CLOSED = 0.0;

    public void init(HardwareMap hw) {
        motor = hw.get(Motor.class, "intakeMotor");
        servo = hw.get(Servo.class, "intakeServo");
        motor.setZeroPowerBehavior(Motor.ZeroPowerBehavior.BRAKE);
        stop();
        close();
    }

    public Action setPower(double power) {
        return () -> { motor.setPower(power); return false; };
    }
    public Action stop() { return setPower(0); }
    public Action intake() { return setPower(INTAKE_POWER); }
    public Action outtake() { return setPower(OUTTAKE_POWER); }
    public Action setServo(double pos) { return () -> { servo.setPosition(pos); return false; }; }
    public Action open() { return setServo(SERVO_OPEN); }
    public Action close() { return setServo(SERVO_CLOSED); }
}
```

## 4. ActionManager.java

```java
package org.firstinspires.ftc.teamcode.action;

import com.example.instantauto.actions.*;
import com.example.instantauto.configs.MetaFieldRegistry;
import com.acmerobotics.roadrunner.*;
import com.acmerobotics.roadrunner.ftc.Actions;
import com.qualcomm.robotcore.hardware.HardwareMap;
import com.qualcomm.robotcore.util.ElapsedTime;
import org.firstinspires.ftc.teamcode.configs.Pose2d;
import org.firstinspires.ftc.teamcode.subsystems.Intake;

import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;

public class ActionManager {
    private MecanumDrive drive;
    private Telemetry telemetry;
    private HardwareMap hw;
    private Intake intake = new Intake();

    public void init(MecanumDrive drive, Telemetry telemetry, HardwareMap hw) {
        this.drive = drive; this.telemetry = telemetry; this.hw = hw;

        // Initialize subsystems
        intake.init(hw);

        // Movement primitives
        UserActionRegistry.register(new MiniAction("STRAFE.TO", this::strafeToFactory));
        UserActionRegistry.register(new MiniAction("SPLINE.TO", this::splineToFactory));

        // Control flow
        UserActionRegistry.register(new MiniAction("WAIT", params -> {
            double[] d = ActionUtils.asDoubles(params, 1);
            return d != null ? ActionUtils.wrap(new SleepAction(d[0])) : null;
        }));

        UserActionRegistry.register(new MiniAction("PARALLEL", params -> {
            List<Action> actions = ActionUtils.asActions(params, drive);
            List<Action> rrActions = actions.stream()
                .map(a -> ActionUtils.adapt(a, telemetry)).collect(Collectors.toList());
            return ActionUtils.wrap(new ParallelAction(rrActions));
        }));

        UserActionRegistry.register(new MiniAction("RACE", params -> {
            List<Action> actions = ActionUtils.asActions(params, drive);
            List<Action> rrActions = actions.stream()
                .map(a -> ActionUtils.adapt(a, telemetry)).collect(Collectors.toList());
            return ActionUtils.wrap(new RaceAction(rrActions));
        }));

        // Subsystem actions
        UserActionRegistry.register(new MiniAction("INTAKE.ON", params -> {
            double[] d = ActionUtils.asDoubles(params, 1);
            return d != null ? intake.setPower(d[0]) : intake.intake();
        }));
        UserActionRegistry.register(new MiniAction("INTAKE.OFF", p -> intake.stop()));
        UserActionRegistry.register(new MiniAction("INTAKE.OUTTAKE", p -> intake.outtake()));
        UserActionRegistry.register(new MiniAction("INTAKE.OPEN", p -> intake.open()));
        UserActionRegistry.register(new MiniAction("INTAKE.CLOSE", p -> intake.close()));

        // Utility
        UserActionRegistry.register(new MiniAction("PRINT", obj -> ActionUtils.wrap(new PrintAction(obj))));
        UserActionRegistry.register(new MiniAction("HELLO.WORLD", p -> ActionUtils.wrap(new PrintAction("Hello World!"))));
    }

    // ========== Factories ==========

    private Action strafeToFactory(Object params) {
        if (params instanceof String) {
            String varName = (String) params;
            MetaFieldRegistry.ConfigEntry<?> entry = MetaFieldRegistry.getEntry(varName);
            if (entry != null && entry.getValue() instanceof Pose2d) {
                final Pose2d p = (Pose2d) entry.getValue();
                return createCachedBuilderAction(builder -> 
                    builder.strafeToSplineHeading(new Vector2d(p.x, p.y), Math.toRadians(p.heading))
                );
            }
        }
        double[] d = ActionUtils.asDoubles(params, 3);
        if (d != null) {
            return createCachedBuilderAction(builder -> 
                builder.strafeToSplineHeading(new Vector2d(d[0], d[1]), Math.toRadians(d[2]))
            );
        }
        return null;
    }

    private Action splineToFactory(Object params) {
        if (params instanceof String) {
            String s = (String) params;
            String[] parts = s.split(",");

            if (parts.length == 5) {
                double[] d = ActionUtils.asDoubles(s, 5);
                if (d != null) {
                    return createCachedBuilderAction(builder -> builder
                        .setTangent(Math.toRadians(d[3]))
                        .splineToSplineHeading(new Pose2d(d[0], d[1], Math.toRadians(d[2])), Math.toRadians(d[4]))
                    );
                }
            }

            if (parts.length == 3) {
                String poseName = parts[0].trim();
                MetaFieldRegistry.ConfigEntry<?> entry = MetaFieldRegistry.getEntry(poseName);
                if (entry != null && entry.getValue() instanceof Pose2d) {
                    Pose2d p = (Pose2d) entry.getValue();
                    try {
                        double startTan = Double.parseDouble(parts[1].trim());
                        double endTan = Double.parseDouble(parts[2].trim());
                        return createCachedBuilderAction(builder -> builder
                            .setTangent(Math.toRadians(startTan))
                            .splineToSplineHeading(new Pose2d(p.x, p.y, Math.toRadians(p.heading)), Math.toRadians(endTan))
                        );
                    } catch (NumberFormatException ignored) {}
                }
            }
        }
        return null;
    }

    private Action createCachedBuilderAction(BuilderAction delegate) {
        return new Action() {
            private Action cachedAction;
            @Override public boolean run() {
                if (cachedAction == null) {
                    cachedAction = delegate.apply(drive.actionBuilder(drive.localizer.getPose())).build();
                }
                return cachedAction.run(new TelemetryPacket());
            }
        };
    }
}
```

## 6. AutonomousBase.java

```java
package org.firstinspires.ftc.teamcode.opmodes;

import com.example.instantauto.actions.AutoParser;
import com.example.instantauto.actions.UserActionRegistry;
import com.example.instantauto.configs.MetaFieldRegistry;
import com.acmerobotics.roadrunner.*;
import com.acmerobotics.roadrunner.ftc.Actions;
import com.qualcomm.robotcore.eventloop.opmode.OpMode;
import org.firstinspires.ftc.teamcode.action.ActionManager;
import org.firstinspires.ftc.teamcode.action.ActionUtils;
import org.firstinspires.ftc.teamcode.configs.ConfigManager;
import org.firstinspires.ftc.teamcode.configs.Pose2d;

import java.io.File;
import java.util.List;
import java.util.stream.Collectors;

public abstract class AutonomousBase extends OpMode {
    protected AutoParser parser;
    protected MecanumDrive drive;
    protected ActionManager actionManager = new ActionManager();
    protected List<Action> actionList = new ArrayList<>();

    @Override
    public void init() {
        // 1. Register Java-side config
        ConfigManager.init(this);

        // 2. Create parser
        parser = new AutoParser("textfiles/GeneralRobotSettings.txt", "textfiles/UserActionSettings.txt");

        // 3. Find and parse active auto (Phase 1: configs only)
        File autoFile = parser.findActiveAutos("textfiles").get(0);
        parser.parseAutoConfig(autoFile);

        // 4. Initialize hardware with Starting pose
        Pose2d startPose = (Pose2d) MetaFieldRegistry.getEntry("Starting").getValue();
        drive = new MecanumDrive(hardwareMap, startPose.getRRPose2d());

        // 5. Register actions (needs MecanumDrive)
        actionManager.init(drive, telemetry, hardwareMap);

        // 6. Set merger for nested action fusion (if/else branches)
        UserActionRegistry.setActionMerger(actions -> ActionUtils.mergeNestedActions(actions, drive));

        // 7. Parse actions (Phase 2: action strings → Action objects)
        parser.parseActions();

        // Telemetry
        telemetry.addLine("Auto: " + MetaFieldRegistry.getEntry("Title").getValue());
        telemetry.addLine("Starting: " + startPose);
        telemetry.addLine("Actions: " + parser.getActions().size());
        for (String err : parser.getActionErrors()) telemetry.addLine("ACT ERR: " + err);
        for (String err : UserActionRegistry.getLoadErrors()) telemetry.addLine("LOAD ERR: " + err);
    }

    @Override
    public void start() {
        // Re-parse with MecanumDrive for trajectory fusion
        List<Action> merged = ActionUtils.asActions(parser.getActionContent(), drive);
        merged = ActionUtils.mergeNestedActions(merged, drive);

        // Adapt to RR Actions
        List<Action> rrActions = merged.stream()
            .map(a -> ActionUtils.adapt(a, telemetry))
            .collect(Collectors.toList());

        // Execute
        Actions.runBlocking(new RaceAction(new SequentialAction(rrActions)));
    }

    @Override
    public void stop() {
        MetaFieldRegistry.clear();
        UserActionRegistry.clear();
    }
}
```

## 8. Execution Flow Summary

```
1. AutonomousBase.init()
   ├─ ConfigManager.init() → registers types, fields, suppliers, conditions
   ├─ AutoParser.parseAutoConfig() → reads Starting pose, overrides
   ├─ MecanumDrive created with Starting pose
   ├─ ActionManager.init() → registers MiniActions (STRAFE.TO, INTAKE.ON, etc.)
   ├─ UserActionRegistry.setActionMerger() → enables if/else fusion
   └─ AutoParser.parseActions() → converts text → Action objects

2. AutonomousBase.start()
   ├─ ActionUtils.asActions() → parses action strings with variable resolution
   ├─ ActionUtils.mergeNestedActions() → fuses trajectories (including if/else branches)
   ├─ ActionUtils.adapt() → wraps each as RR Action
   └─ Actions.runBlocking(RaceAction(SequentialAction(...))) → executes
```
